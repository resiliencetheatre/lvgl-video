cmake_minimum_required(VERSION 3.18)
project(lvgl_video VERSION 0.2.0 LANGUAGES C CXX ASM)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_VIDEO_BACKEND "fbdev" CACHE STRING "Display backend: wayland or fbdev")
set_property(CACHE LVGL_VIDEO_BACKEND PROPERTY STRINGS wayland fbdev)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl" CACHE PATH "LVGL source directory")
if(NOT LVGL_VIDEO_BACKEND MATCHES "^(wayland|fbdev)$")
    message(FATAL_ERROR "LVGL_VIDEO_BACKEND must be wayland or fbdev")
endif()
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET
    gstreamer-app-1.0>=1.10 gstreamer-video-1.0 gio-2.0)

# Use upstream's build without installing its private library/headers or demos.
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(LV_BUILD_USE_KCONFIG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_SET_CONFIG_OPTS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC
    APP_USE_WAYLAND=$<STREQUAL:${LVGL_VIDEO_BACKEND},wayland>)
target_link_libraries(lvgl PUBLIC Threads::Threads m)
if(LVGL_VIDEO_BACKEND STREQUAL "wayland")
    include(cmake/Wayland.cmake)
endif()
add_executable(lvgl-video src/main.c src/display.c src/ui.c src/options.c src/session.c src/video.c)
target_link_libraries(lvgl-video PRIVATE lvgl::lvgl PkgConfig::GSTREAMER)
target_compile_options(lvgl-video PRIVATE -Wall -Wextra)
install(TARGETS lvgl-video RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

option(LVGL_VIDEO_TESTS "Build native GTK Pipe transport interoperability tests" OFF)
if(LVGL_VIDEO_TESTS)
    enable_testing()
    set(GTK_PIPE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../gtk-pipe" CACHE PATH "GTK Pipe reference checkout")
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
    add_executable(test-transport tests/test_transport.c tests/gtk-reference.c src/session.c src/video.c)
    target_include_directories(test-transport PRIVATE src)
    target_compile_definitions(test-transport PRIVATE GTK_PIPE_SOURCE="${GTK_PIPE_SOURCE_DIR}/gtk-pipe.c")
    target_compile_options(test-transport PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
    target_link_options(test-transport PRIVATE -Wl,--gc-sections)
    target_link_libraries(test-transport PRIVATE lvgl::lvgl PkgConfig::GSTREAMER PkgConfig::GTK)
    add_executable(test-options tests/test_options.c src/options.c)
    target_include_directories(test-options PRIVATE src)
    target_compile_options(test-options PRIVATE -Wall -Wextra)
    target_link_libraries(test-options PRIVATE PkgConfig::GSTREAMER)
    add_test(NAME options COMMAND test-options)
    add_test(NAME transport COMMAND test-transport)
    add_test(NAME transport-no-aec COMMAND test-transport --disable-echo-cancellation)
    add_test(NAME transport-usb-format COMMAND test-transport --usb)
    set_tests_properties(transport transport-no-aec transport-usb-format PROPERTIES TIMEOUT 20)
endif()
