cmake_minimum_required(VERSION 3.18)
project(lvgl_com VERSION 0.2.0 LANGUAGES C CXX ASM)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_COM_BACKEND "wayland" CACHE STRING "Display backend: wayland or fbdev")
set_property(CACHE LVGL_COM_BACKEND PROPERTY STRINGS wayland fbdev)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl" CACHE PATH "LVGL source directory")
set(LVGL_COM_ASSET_DIR "${CMAKE_INSTALL_FULL_DATADIR}/lvgl-com" CACHE PATH "Runtime asset directory")
if(NOT LVGL_COM_BACKEND MATCHES "^(wayland|fbdev)$")
    message(FATAL_ERROR "LVGL_COM_BACKEND must be wayland or fbdev")
endif()
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
find_package(ZLIB REQUIRED)

# Use upstream's build without installing its private library/headers or demos.
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(LV_BUILD_USE_KCONFIG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_SET_CONFIG_OPTS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC
    APP_USE_WAYLAND=$<STREQUAL:${LVGL_COM_BACKEND},wayland>
    APP_ASSET_DIR="${LVGL_COM_ASSET_DIR}")
target_link_libraries(lvgl PUBLIC PNG::PNG ZLIB::ZLIB Threads::Threads m)
if(LVGL_COM_BACKEND STREQUAL "wayland")
    include(cmake/Wayland.cmake)
endif()
add_executable(lvgl-com src/main.c src/display.c src/mini.c src/log.c)
target_link_libraries(lvgl-com PRIVATE lvgl::lvgl)
target_compile_options(lvgl-com PRIVATE -Wall -Wextra)
install(TARGETS lvgl-com RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/link.png DESTINATION ${CMAKE_INSTALL_DATADIR}/lvgl-com)
